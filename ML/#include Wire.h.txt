#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_BME680.h>
#include <Adafruit_SHT31.h>
#include <HardwareSerial.h>
#include <LiquidCrystal_I2C.h>

#define SDA_PIN 21
#define SCL_PIN 22
#define RX_PIN 16
#define TX_PIN 17

#define MIN_TIME 120000   // 2 minutes
#define SCROLL_INTERVAL 3000 // 3 seconds per scroll

Adafruit_BME680 bme;
Adafruit_SHT31 sht31 = Adafruit_SHT31();
HardwareSerial MHZSerial(2);

LiquidCrystal_I2C lcd(0x27,16,2);

byte cmd_getData[9] = {0xFF,0x01,0x86,0,0,0,0,0,0x79};
byte response[9];

unsigned long startTime;
bool frozen = false;
int scrollStep = 0;

// Previous readings for stability check
float prevTemp=0, prevPressure=0, prevGas=0, prevHumidity=0;
int prevCO2=0;

// Forward declaration
void displayScroll(float temp,float humidity,float pressure,float gas,int co2);

void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN,SCL_PIN);

  // BME680 init
  if(!bme.begin(0x76)) while(1);
  bme.setTemperatureOversampling(BME680_OS_NONE);
  bme.setHumidityOversampling(BME680_OS_NONE);
  bme.setPressureOversampling(BME680_OS_4X);
  bme.setGasHeater(320,150);

  // SHT35 init
  if(!sht31.begin(0x44)) while(1);

  // MH-Z14A init
  MHZSerial.begin(9600,SERIAL_8N1,RX_PIN,TX_PIN);

  // LCD init
  lcd.begin();
  lcd.backlight();

  startTime = millis();
}

void loop() {

  float temp = sht31.readTemperature();
  float humidity = sht31.readHumidity();
  float pressure=0;
  float gas=0;
  int co2=0;

  // BME680 reading
  if(bme.performReading()){
    pressure = bme.pressure/100.0;
    gas = bme.gas_resistance/1000.0;
  }

  // MH-Z14A reading
  while(MHZSerial.available()) MHZSerial.read();
  MHZSerial.write(cmd_getData,9);
  delay(100);
  if(MHZSerial.available()>=9){
    for(int i=0;i<9;i++) response[i]=MHZSerial.read();
    if(response[0]==0xFF && response[1]==0x86)
      co2=(response[2]<<8)+response[3];
  }

  // ===== Stabilization check =====
  float diffTemp=abs(temp-prevTemp);
  float diffHumidity=abs(humidity-prevHumidity);
  float diffPressure=abs(pressure-prevPressure);
  float diffGas=abs(gas-prevGas);
  int diffCO2=abs(co2-prevCO2);

  bool stable = (diffTemp<0.2) &&
                (diffHumidity<0.5) &&
                (diffPressure<0.5) &&
                (diffGas<0.2) &&
                (diffCO2<10);

  if(!frozen && (millis()-startTime<MIN_TIME || !stable)){
    // ===== Stabilizing phase =====
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("Stabilizing...");
    lcd.setCursor(0,1);
    lcd.print((millis()-startTime)/1000);
    lcd.print(" sec");
    delay(500); // update faster for countdown
  } else {
    // ===== Frozen phase =====
    frozen = true;

    // display scrolling readings
    displayScroll(temp,humidity,pressure,gas,co2);

    scrollStep = (scrollStep+1)%3;
    delay(SCROLL_INTERVAL);
  }

  // Save previous readings
  prevTemp=temp;
  prevHumidity=humidity;
  prevPressure=pressure;
  prevGas=gas;
  prevCO2=co2;
}

// ===== Scroll display function =====
void displayScroll(float temp,float humidity,float pressure,float gas,int co2){
  lcd.clear();
  switch(scrollStep){
    case 0: // Pressure + Temp
      lcd.setCursor(0,0);
      lcd.print("P:");
      lcd.print(pressure,1);
      lcd.setCursor(9,0);
      lcd.print("T:");
      lcd.print(temp,1);
      lcd.write(223); // degree symbol
      lcd.print("C");
      break;
    case 1: // Humidity + Gas
      lcd.setCursor(0,0);
      lcd.print("H:");
      lcd.print(humidity,1);
      lcd.setCursor(9,0);
      lcd.print("G:");
      lcd.print(gas,2);
      break;
    case 2: // CO2
      lcd.setCursor(0,0);
      lcd.print("CO2:");
      lcd.print(co2);
      lcd.print(" ppm");
      break;
  }
}
